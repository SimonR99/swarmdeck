cmake_minimum_required(VERSION 3.16)
project(swarmdeck_ultra_fusion_numerics LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(ultra_fusion_numerics src/numerics.cpp)
target_include_directories(ultra_fusion_numerics PUBLIC include)
target_link_libraries(ultra_fusion_numerics PUBLIC Eigen3::Eigen)
target_compile_features(ultra_fusion_numerics PUBLIC cxx_std_17)
target_compile_options(ultra_fusion_numerics PRIVATE -Wall -Wextra -Wpedantic)
# Deliberately no x86 ISA flags or host-specific -march=native.
include(CTest)
if(BUILD_TESTING)
  add_executable(numerics_test tests/numerics_test.cpp)
  target_link_libraries(numerics_test PRIVATE ultra_fusion_numerics)
  add_test(NAME numerics COMMAND numerics_test)
endif()
